## 2. 基础框架设置

### 2.1 app.js 全局配置

```javascript
// app.js
App({
  onLaunch: function() {
    // 初始化云开发环境
    if (!wx.cloud) {
      console.error('请使用 2.2.3 或以上的基础库以使用云能力');
    } else {
      wx.cloud.init({
        env: 'your-env-id',
        traceUser: true,
      });
    }

    // 获取用户信息
    this.getUserInfo();
    
    // 初始化全局数据
    this.globalData = {
      userInfo: null,
      isLogin: false,
      theme: 'light',
      tasks: [],
    };
  },

  // 获取用户信息
  getUserInfo: function() {
    // 调用登录云函数
    wx.cloud.callFunction({
      name: 'login',
      success: res => {
        this.globalData.userInfo = res.result.userInfo;
        this.globalData.isLogin = true;
      },
      fail: err => {
        console.error('[云函数] [login] 调用失败', err);
      }
    });
  },

  // 全局数据更新方法
  updateGlobalData: function(key, value) {
    this.globalData[key] = value;
  }
});
```

### 3.1 单位使用规范

    // 事件处理逻辑
  },
  
  // 页面导航
  navigateToPage: function(page, params) {
    wx.navigateTo({
      url: `/${page}?${this.formatParams(params)}`
    });
  },
  
  // 格式化参数
  formatParams: function(params) {
    return Object.keys(params)
      .map(key => `${key}=${encodeURIComponent(params[key])}`)
      .join('&');
  }
});

### 5.2 左滑操作组件示例

```javascript
// components/swipe-action/swipe-action.js
Component({
  properties: {
    // 是否开启左滑
    disabled: {
      type: Boolean,
      value: false
    },
    // 按钮配置
    buttons: {
      type: Array,
      value: [
        {
          text: '编辑',
          style: 'background-color: #FFB79D; color: #fff;'
        },
        {
          text: '删除',
          style: 'background-color: #FF9E80; color: #fff;'
        }
      ]
    }
  },
  
  data: {
    // 滑动距离
    translateX: 0,
    // 是否正在滑动
    sliding: false,
    // 是否打开
    opened: false
  },
  
  methods: {
    // 触摸开始
    onTouchStart: function(e) {
      if (this.data.disabled) return;
      this.startX = e.touches[0].clientX;
    },
    
    // 触摸移动
    onTouchMove: function(e) {
      if (this.data.disabled) return;
      
      const moveX = e.touches[0].clientX;
      let translateX = moveX - this.startX;
      
      // 限制只能左滑，且有最大滑动距离
      const maxTranslateX = -200; // 按钮总宽度
      translateX = Math.max(translateX, maxTranslateX);
      translateX = Math.min(translateX, 0);
      
      this.setData({
        translateX: translateX,
        sliding: true
      });
    },
    
    // 触摸结束
    onTouchEnd: function() {
      if (this.data.disabled) return;
      
      let translateX = this.data.translateX;
      let opened = this.data.opened;
      
      // 判断是否需要打开或关闭
      if (translateX < -100) {
        translateX = -200; // 完全打开
        opened = true;
      } else {
        translateX = 0; // 关闭
        opened = false;
      }
      
      this.setData({
        translateX: translateX,
        sliding: false,
        opened: opened
      });
    },
    
    // 点击按钮
    onButtonTap: function(e) {
      const index = e.currentTarget.dataset.index;
      const button = this.data.buttons[index];
      
      // 触发按钮点击事件
      this.triggerEvent('buttontap', { button, index });
      
      // 如果不是删除按钮，点击后关闭
      if (button.text !== '删除') {
        this.close();
      }
    },
    
    // 关闭滑动
    close: function() {
      this.setData({
        translateX: 0,
        opened: false
      });
    }
  }
});
```

```javascript
// utils/request.js
const request = {
  // 封装云函数调用
  call: function(name, data = {}) {
    return new Promise((resolve, reject) => {
      wx.cloud.callFunction({
        name: name,
        data: data,
        success: res => {
          resolve(res.result);
        },
        fail: err => {
          console.error(`[云函数] [${name}] 调用失败`, err);
          wx.showToast({
            title: '网络请求失败',
            icon: 'none'
          });
          reject(err);
        }
      });
    });
  },
  
  // 封装数据库操作
  db: function() {
    return wx.cloud.database();
  },
  
  // 显示加载提示
  showLoading: function(title = '加载中...') {
    wx.showLoading({
      title: title,
      mask: true
    });
  },
  
  // 隐藏加载提示
  hideLoading: function() {
    wx.hideLoading();
  }
};

export default request;
```

### 6.2 日期处理函数

```javascript
// utils/date.js
const dateUtil = {
  // 格式化日期
  formatDate: function(date, format = 'YYYY-MM-DD') {
    const d = new Date(date);
    const year = d.getFullYear();
    const month = String(d.getMonth() + 1).padStart(2, '0');
    const day = String(d.getDate()).padStart(2, '0');
    const hours = String(d.getHours()).padStart(2, '0');
    const minutes = String(d.getMinutes()).padStart(2, '0');
    
    return format
      .replace('YYYY', year)
      .replace('MM', month)
      .replace('DD', day)
      .replace('HH', hours)
      .replace('mm', minutes);
  },
  
  // 获取今天的日期
  getToday: function() {
    return new Date(new Date().setHours(0, 0, 0, 0));
  },
  
  // 获取星期几
  getWeekday: function(date) {
    const weekdays = ['日', '一', '二', '三', '四', '五', '六'];
    const d = new Date(date);
    return weekdays[d.getDay()];
  },
  
  // 计算下次循环日期
  calculateNextDate: function(currentDate, cycleType, cycleParams) {
    const d = new Date(currentDate);
    
    switch (cycleType) {
      case 'daily':
        d.setDate(d.getDate() + 1);
        break;
      case 'weekly':
        // 找到下一个指定的星期几
        const currentWeekday = d.getDay();
        const targetWeekdays = cycleParams.weekdays || [1];
        let minDiff = 7;
        
        for (const day of targetWeekdays) {
          let diff = day - currentWeekday;
          if (diff <= 0) diff += 7;
          minDiff = Math.min(minDiff, diff);
        }
        
        d.setDate(d.getDate() + minDiff);
        break;
      case 'monthly':
        // 设置为下个月的指定日期
        d.setMonth(d.getMonth() + 1);
        const targetDay = cycleParams.dayOfMonth || 1;
        // 处理月末日期
        const lastDay = new Date(d.getFullYear(), d.getMonth() + 1, 0).getDate();
        d.setDate(Math.min(targetDay, lastDay));
        break;
    }
    
    return d;
  }
};

export default dateUtil;
```

## 7. 性能优化策略

### 7.1 代码优化
- 使用按需加载，减少初始加载资源
- 避免不必要的重渲染
- 合理使用缓存

### 7.2 图片优化
- 使用合适尺寸的图片
- 图片懒加载
- 图片压缩

### 7.3 网络请求优化
- 批量请求合并
- 使用缓存减少请求
- 请求失败重试机制

### 7.4 渲染性能优化
- 使用虚拟列表处理长列表
- 减少不必要的节点嵌套
- 避免频繁的页面切换
